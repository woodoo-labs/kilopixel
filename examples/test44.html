Mathematical Crescent Moon

By splitting the visualization into two layers, we can use HTML5 Canvas Compositing to perfectly erase the inner circle, while keeping a faint wireframe outline visible in the background layer to prove the geometry.

<pxl-stage ratio="5 / 3">
  <!-- Define primary radius and desired thickness (a) -->
  <pxl-var id="moonR1" value="200"></pxl-var>
  <pxl-var id="moonA" value="100"></pxl-var>
  <pxl-var id="moonBlend" value="'destination-out'"></pxl-var>

  <!-- Compute exact center offset (dx) and radius (r2) -->
  <pxl-var id="moonDx" value="(ref.moonA.value**2 - ref.moonR1.value**2) / (2 * ref.moonA.value)"></pxl-var>
  <pxl-var id="moonR2" value="ref.moonA.value - ref.moonDx.value"></pxl-var>

  <!-- Layer 1: Faint wireframe background to see what's happening -->
  <pxl-layer x="500" y="300">
    <pxl-circle r="ref.moonR1.value" stroke="#cbd5e1" fill="none"></pxl-circle>
    <pxl-circle x="ref.moonDx.value" r="ref.moonR2.value" stroke="#cbd5e1" fill="none"></pxl-circle>
  </pxl-layer>

  <!-- Layer 2: The Boolean composite with a glowing drop shadow! -->
  <pxl-layer x="500" y="300" filter="dropShadow(0, 0, 15, '#f59e0b')">
    <!-- Draw Solid Moon -->
    <pxl-circle r="ref.moonR1.value" fill="#fcd34d"></pxl-circle>
    
    <!-- ERASER: Overlapping pixels are erased due to blend mode -->
    <pxl-circle x="ref.moonDx.value" r="ref.moonR2.value" fill="#000" blend="ref.moonBlend.value"></pxl-circle>
  </pxl-layer>
</pxl-stage>
